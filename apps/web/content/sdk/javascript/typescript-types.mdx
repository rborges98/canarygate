## `CanaryGate`

The main SDK class.

```ts
class CanaryGate {
  constructor(apiKey: string, options?: CanaryGateOptions)
  init(): Promise<void>
  getFlag(key: string): FlagData | undefined
  getFlags(): FlagData[]
  isStale(): boolean
  getLastSyncAt(): Date | null
  disconnect(): void
}
```

## `CanaryGateOptions`

```ts
interface CanaryGateOptions {
  /** User ID for deterministic rollout hashing. */
  userId?: string
  /** API base URL. Default: 'https://api.canarygate.io' */
  baseUrl?: string
  /** Time in ms before considering the SDK stale. Default: 60000 */
  staleThreshold?: number
}
```

## `FlagData`

Union type representing any flag.

```ts
type FlagData = BooleanFlagData | RolloutFlagData
```

## `BooleanFlagData`

```ts
interface BooleanFlagData {
  key: string
  type: 'boolean'
  enabled: boolean
  description: string | null
}
```

## `RolloutFlagData`

```ts
interface RolloutFlagData {
  key: string
  type: 'rollout'
  enabled: boolean
  percentage: number
  description: string | null
}
```

## Narrowing by type

Use `type` for safe narrowing:

```ts
const flag = canary.getFlag('new-dashboard')

if (flag?.type === 'rollout') {
  // TypeScript knows flag is RolloutFlagData here
  console.log(flag.percentage)
}
```

## Full exports

```ts
import {
  CanaryGate,
```
